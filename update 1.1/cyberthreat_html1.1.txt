    <script>
        setInterval(() => {
            document.getElementById('clock').innerText = new Date().toISOString().replace('T', ' ').substring(0, 19);
        }, 1000);

        async function triggerQuantumScan() {
            const targetVal = document.getElementById('targetInput').value.trim();
            if(!targetVal) return;

            const consoleBox = document.getElementById('tickerConsole');
            consoleBox.innerHTML = "";

            const fakeLogs = [
                { text: `> [CONNECT] Initiating deployment interface tunnel to ${targetVal}...`, c: "" },
                { text: `> [OSINT] Running parallel query blocks across 4 public intelligence databases...`, c: "ticker-success" },
                { text: `> [PORT SCAN] Checking exposed service architectures...`, c: "" },
                { text: `> [AUDIT] Analyzing baseline response security header configurations...`, c: "ticker-warn" }
            ];

            for (let i = 0; i < fakeLogs.length; i++) {
                await new Promise(res => setTimeout(res, 200));
                const d = document.createElement('div');
                d.className = `ticker-line ${fakeLogs[i].c}`;
                d.innerText = fakeLogs[i].text;
                consoleBox.appendChild(d);
                consoleBox.scrollTop = consoleBox.scrollHeight;
            }

            try {
                // Hitting the matching 127.0.0.1 link loopback line explicitly
                const response = await fetch("http://127.0.0.1:8000/analyze/target", {
                    method: "POST",
                    headers: { "Content-Type": "application/json" },
                    body: JSON.stringify({ target: targetVal })
                });

                if(!response.ok) throw new Error();
                const data = await response.json();

                document.getElementById('dashboardCore').classList.remove('hidden');
                document.getElementById('riskScore').innerText = `${data.overall_risk_score}/100 (${data.overall_severity})`;
                document.getElementById('honeypotProb').innerText = `${data.predictive_analytics.honeypot_probability * 100}%`;
                document.getElementById('timeToBreach').innerText = data.predictive_analytics.estimated_time_to_compromise;
                document.getElementById('threatActor').innerText = data.predictive_analytics.primary_threat_actor_match;
                document.getElementById('nodeGateway').innerText = `IP Node: ${data.target}`;

                const listContainer = document.getElementById('vulnList');
                listContainer.innerHTML = "";

                data.open_ports.forEach(p => {
                    const el = document.createElement('div');
                    el.className = "data-item";
                    el.innerHTML = `<span style="color:var(--crimson-red)">[PORT]</span> <strong>${p.service} (${p.port})</strong> is exposed.<br><small style="color:var(--text-muted)">${p.risk_note}</small>`;
                    listContainer.appendChild(el);
                });

                data.header_audit.forEach(h => {
                    if(!h.present) {
                        const el = document.createElement('div');
                        el.className = "data-item";
                        el.innerHTML = `<span style="color:var(--warning-amber)">[MISCONFIG]</span> Missing security header: <strong>${h.header}</strong>.<br><small style="color:var(--text-muted)">Fix: ${h.recommendation}</small>`;
                        listContainer.appendChild(el);
                    }
                });

                const s = document.createElement('div');
                s.className = "ticker-line ticker-success";
                s.innerText = `> [SUCCESS] Telemetry stream generated and parsed perfectly in ${data.analysis_duration_ms}ms. Matrix stable.`;
                consoleBox.appendChild(s);
                consoleBox.scrollTop = consoleBox.scrollHeight;

            } catch (err) {
                const e = document.createElement('div');
                e.className = "ticker-line ticker-warn";
                e.innerText = `> [ERROR] Interface dropped connection lane to backend. Verify your terminal execution state.`;
                consoleBox.appendChild(e);
            }
        }
    </script>
